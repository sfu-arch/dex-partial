================================================================================
ARG REFERENCE (newbench_latency positions)
================================================================================

pos  1  = nodecount       = 2       (1 memory + 1 compute)
pos  2  = read%           = 100
pos  3  = insert%         = 0
pos  4  = update%         = 0
pos  5  = delete%         = 0
pos  6  = range%          = 0
pos  7  = threads         = 30      (client threads)
pos  8  = mem_threads     = 4
pos  9  = cache_mb        = 32/64/128/256/512   ← THE KEY VARIABLE
pos 10  = uniform         = 1 (uniform) or 0 (zipf)
pos 11  = theta           = 0.0 (uniform) or 0.99 (zipf)
pos 12  = bulk_load       = 50      (50M keys)
pos 13  = warmup          = 10      (10M ops, not measured)
pos 14  = op_num          = 50      (50M measured ops)
pos 15  = check           = 0
pos 16  = time_based      = 1
pos 17  = early_stop      = 0       (disabled — full run)
pos 18  = index           = 0       (DEX = cachepush)
pos 19  = rpc_rate        = 0       (one-sided RDMA only)
pos 20  = admission_rate  = 0.1     (10% random admission)
pos 21  = auto_tune       = 0
pos 22  = kMaxThread      = 30
